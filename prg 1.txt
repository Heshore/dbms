-- Foreign key creation
CREATE TABLE loandetails(
    loanid INT PRIMARY KEY,
    accno INT,
    loanamt INT,
    FOREIGN KEY (accno) REFERENCES bank(accno)
);

DESC loandetails;

-- Combining two tables using JOIN
SELECT b.cust_name, l.loanid, l.loanamt
FROM bank b
INNER JOIN loandetails l
ON b.accno = l.accno;
